using System; using System.Collections.Generic; using
System.Xml.Serialization;

#region Response Models

[XmlRoot(“VerifyAcctNumberInqRs”)] public class VerifyAcctNumberInqRs {
[XmlElement(“MsgRsHdr”)] public MsgRsHdrType? MsgRsHdr { get; set; }

    // Based on debugger screenshots showing RespDisplayData = string[4]
    [XmlElement("RespDisplayData")]
    public string[]? RespDisplayData { get; set; }

}

public class MsgRsHdrType { [XmlElement(“RqUID”)] public string? RqUID {
get; set; }

    [XmlElement("TransactionTime")]
    public string? TransactionTime { get; set; }

    [XmlElement("ResponseCode")]
    public int ResponseCode { get; set; }

    [XmlElement("Status")]
    public StatusType? Status { get; set; }

}

public class StatusType { [XmlElement(“Severity”)] public int Severity {
get; set; }

    [XmlElement("StatusCode")]
    public int StatusCode { get; set; }

    [XmlElement("StatusDesc")]
    public string? StatusDesc { get; set; }

}

#endregion

#region DTO

public class AccountVerificationResponseDTO { public string? MemberName
{ get; set; }

    public string? AccountStatus { get; set; }

    public List<string> Alerts { get; set; } = new();

    public string? StatusDescription { get; set; }

    public int ResponseCode { get; set; }

}

#endregion

#region Mapper

public static class VerifyAcctEntityMapper { public static
AccountVerificationResponseDTO MapVerifyAcct( VerifyAcctNumberInqRs
response) { if (response == null) return new
AccountVerificationResponseDTO();

        var dto = new AccountVerificationResponseDTO
        {
            ResponseCode = response.MsgRsHdr?.ResponseCode ?? 0,
            StatusDescription = response.MsgRsHdr?.Status?.StatusDesc
        };

        if (response.RespDisplayData == null)
            return dto;

        foreach (var item in response.RespDisplayData)
        {
            if (item.StartsWith("Member Name:",
                StringComparison.OrdinalIgnoreCase))
            {
                dto.MemberName =
                    item.Replace("Member Name:", "").Trim();
            }
            else if (item.StartsWith("Account Status:",
                StringComparison.OrdinalIgnoreCase))
            {
                dto.AccountStatus =
                    item.Replace("Account Status:", "").Trim();
            }
            else if (!item.StartsWith("Member Alerts:",
                StringComparison.OrdinalIgnoreCase))
            {
                dto.Alerts.Add(item);
            }
        }

        return dto;
    }

}

#endregion
