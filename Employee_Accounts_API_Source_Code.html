api_response.py

Responsibility: Connects to the Employee Accounts REST API, sends the request payload, receives the JSON response, and returns it to the caller.

# api_response.py

import http.client
import json


def get_employee_accounts():
    conn = http.client.HTTPSConnection("intg-dpa-lb.nfcutest.net", 5508)

    payload = json.dumps({
        "rquid": "b1f8570b-b2c7-4789-88ce-6767757b2857",
        "employeeID": "37015"
    })

    headers = {
        "Content-Type": "application/json"
    }

    conn.request(
        "POST",
        "/NFCUBranchTellerServices/rest/employee-account-relationships/v1/employee/relationships/accounts",
        payload,
        headers
    )

    res = conn.getresponse()
    data = res.read()

    return data.decode("utf-8")

main.py

Responsibility: Calls the API function, parses the JSON response, extracts the accounts collection, and displays account information.

# main.py

from api_response import get_employee_accounts
import json

response = get_employee_accounts()
response_json = json.loads(response)

accounts = response_json["accounts"]

print(f"Total accounts: {len(accounts)}")

for account in accounts:
    print(account["accountType"], account["accountNumber"])
